# JAKA ZU Script

#变量声明

Z = 215		#触纸高度

vel1 = 30		#速度1
acc1 =90	   	#加速度1

vel2 = 60		#速度2
acc2 =180	#加速度2

tol = 1		 #转弯半径

#平面变换，偏移
dy = 0 			
dz = 100 	  
drx = 0 

#======================点位===================
#机器人原点
Home = [0, 0, -90, 0, -90, 0]

#写字某一段轨迹点位
P1 = [-67.771302, -296.384003, Z, 180, -0, 180]
P2 = [80.996697, -296.384, Z, 180, -0, 180]
P3 = [80.996699, -257.679998, Z, 180, -0, 180]
P4 = [165.412699, -257.680002, Z, 180, -0, 180]

P5 = [165.412698, -170.975999, Z, 180, -0, 180] 
P6 = [86.548701, -170.976002, Z, 180, -0, 180]
P7 = [86.548701, -203.657699, Z, 180, -0, 180]
P8 = [-44.779297, -193.593703, Z, 180, -0, 180]
P9 = [-90.539299, -229.7673, Z, 180, -0, 180]

#======================点位===================

#======================轨迹===================

#声明坐标系
set_tool_id(1) 
set_user_frame_id(1) 

#关节运动到原点
movj(Home, 0, vel2, acc2, 0)

P0 =yz_plane_trans(P1, dy, dz, drx)  #落笔
movl(P0, 0, vel2, acc2, 0) 

movl(P1, 0, vel1, acc1, 0)
movl(P2, 0, vel1, acc1, tol) 
movl(P3, 0, vel1, acc1, tol) 
movl(P4, 0, vel1, acc1, tol)   

P0 =yz_plane_trans(P4, dy, dz, drx)  #抬笔
movl(P0, 0, vel2, acc2, 0)

P0 =yz_plane_trans(P5, dy, dz, drx)  #落笔
movl(P0, 0, vel2, acc2, 0)

movl(P5, 0, vel1, acc1, 0)
movl(P6, 0, vel1, acc1, tol) 
movl(P7, 0, vel1, acc1, tol) 
movl(P8, 0, vel1, acc1, tol)   
movl(P9, 0, vel1, acc1, tol)  

P0 =yz_plane_trans(P9, dy, dz, drx)  #抬笔
movl(P0, 0, vel2, acc2, 0)

#关节运动到原点
movj(Home, 0, vel2, acc2, 0)
